cmake_minimum_required(VERSION 3.16)
project(MegaSaturn LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
add_library(megasaturn_core pc/megasaturn_cpu.c pc/megasaturn_bus.c pc/megasaturn_exec.c)
target_include_directories(megasaturn_core PUBLIC pc)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
  target_compile_options(megasaturn_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
foreach(test IN ITEMS cpu bus exec move move_v05 subroutine_v06 move_address)
  add_executable(test_${test} pc/test_${test}.c)
  target_link_libraries(test_${test} PRIVATE megasaturn_core)
  add_test(NAME ${test} COMMAND test_${test})
endforeach()
